    )
}

## Kepler 方程式の根の一括検証


次のセルは Kepler 方程式 $f(E) = E - e\sin E - M = 0$ の根についての七つの提案を、`ScalarRootVerifier` の一括検証で確かめる。
方程式の導出と解法の比較は [Kepler 方程式と離心率](../../../scioxide-rootfind/examples/kepler-equation/) にあり、既定値も同じ $e = 0.6$、$M = 1.2$ rad にそろえる。

値の拡張は正弦の包含区間を `hull()` で一つの区間にまとめて作り、導関数 $f'(E) = 1 - e\cos E$ も同じように作る。
$e$ と $M$ は入力の binary64 値そのものを厳密な定数とする。
提案の中心 $E^*$ は f64 の固有メソッドによる Newton 法で求めた近似値で、検証の対象ではない。
提案は全域 $[0, 2\pi]$ での一意性、$E^*$ を中心とする半幅 $10^{-3}$・$10^{-9}$・$10^{-15}$・0 の区間での一意性、根を含まない区間と含む区間での不在である。
作業領域は `ProofRequest::proofs(7)` で事前確保し、検証器のバイト上限を 1 MiB にする。

作業領域の要求の表は丸めを使わないので、どの環境でも出る。
判定と確立した事実の列はネイティブ環境だけで埋まり、wasm32 では「未実行」と表示する。

```rust
//| id: validated-api-kepler-proof
//| ref: validated-api-kepler-proof
//| title: Kepler 方程式の根を区間 Newton 法の一括検証で含む区間を求める
//| run: button
//| uses: [validated-api-environment]
//| crates: [scioxide-core, scioxide-validated]
//| inputs:
//|   eccentricity: {"type":"range","label":"離心率 e [-]","min":0,"max":0.99,"step":0.01,"value":0.6}
//|   mean: {"type":"range","label":"平均近点角 M [rad]","min":0,"max":6.28,"step":0.01,"value":1.2}
//| cases:
//|   - {"id":"circular","label":"e = 0（円軌道）","description":"根は E* = M = 1.2 そのもので、f は厳密に一次式になる。半幅 0 の点の区間でも一意性が Proved になる。","inputs":{"eccentricity":0}}
//|   - {"id":"near-two-pi","label":"M = 6.28（2π の直前）","description":"E* ≈ 6.275222 では 1 ulp が約 8.9×10⁻¹⁶ なので、半幅 10⁻¹⁵ の区間は端点の丸めで証明できず Inconclusive(Reason(Inequality)) になる。","inputs":{"mean":6.28}}
//|   - {"id":"periapsis","label":"e = 0.99、M = 0（近点）","description":"根は E* = 0 で点の区間でも Proved になるが、[0.01, 0.1] での不在は x が二回現れる依存性で包含区間が 0 を含み、Inconclusive(Reason(Inequality)) になる。","inputs":{"eccentricity":0.99,"mean":0}}
//| tests:
//|   - case: default
//|     expect:
//|       - {"artifact":"table-0","path":"/rows/0/1","equals":7}
//|       - {"artifact":"table-0","path":"/rows/1/1","range":{"min":1000,"max":20000}}
//|       - {"artifact":"table-0","path":"/rows/2/1","equals":1048576}
//|       - {"artifact":"table-1","path":"/rows/0/0","equals":"Unique"}
//|       - {"artifact":"table-1","path":"/rows/1/2","approx":{"value":1.785141603274313,"atol":1e-12}}
//|       - {"artifact":"table-1","path":"/rows/4/2","approx":{"value":1.786141603274313,"atol":1e-12}}
//|       - {"artifact":"table-1","path":"/rows/5/0","equals":"Empty"}
//|       - {"artifact":"table-1","path":"/rows/5/3","approx":{"value":1.886141603274313,"atol":1e-12}}
//|   - case: circular
//|     expect:
//|       - {"artifact":"table-1","path":"/rows/4/2","approx":{"value":1.2,"atol":1e-15}}
//|       - {"artifact":"table-1","path":"/rows/4/3","approx":{"value":1.2,"atol":1e-15}}
//|   - case: near-two-pi
//|     expect:
//|       - {"artifact":"table-1","path":"/rows/3/2","approx":{"value":6.275222165469160,"atol":1e-12}}
//|   - case: periapsis
//|     expect:
//|       - {"artifact":"table-1","path":"/rows/4/2","approx":{"value":0,"atol":1e-15}}
//|       - {"artifact":"table-1","path":"/rows/5/2","approx":{"value":0.01,"atol":1e-15}}
use scioxide_core::{ExecutionContext, ExtendedInterval, SciOxideError};
use scioxide_validated::{
    DecoratedInterval as D, ProofRequest, RootClaim, RootVerificationReport,
    ScalarRootCertificate, ScalarRootVerifier, ValidatedOptions,
};

// 1. f(E) = E − e sin E − M と f'(E) = 1 − e cos E の区間拡張（e と M は入力の binary64 値）
let ecc = D::point(eccentricity).map_err(|e| e.to_string())?;
let anomaly = D::point(mean).map_err(|e| e.to_string())?;
let one = D::point(1.0).map_err(|e| e.to_string())?;
let value = |x: D<f64>| x.sub(&ecc.mul(&x.sin()?.hull())?)?.sub(&anomaly);
let derivative = |x: D<f64>| one.sub(&ecc.mul(&x.cos()?.hull())?);
// 2. 提案の中心 E*: f64 の固有メソッドによる Newton 法（E₀ = π）。検証されない近似値である。
let mut center = std::f64::consts::PI;
for _ in 0..60 {
    center -=
        (center - eccentricity * center.sin() - mean) / (1.0 - eccentricity * center.cos());
}
// 3. 主張と区間。半幅 0 は点の区間、最後の二つは根を含まない区間と含む区間での不在の主張。
let tau = std::f64::consts::TAU;
let proposals = [
    (RootClaim::Unique, 0.0, tau, "[0, 2π]"),
    (RootClaim::Unique, center - 1e-3, center + 1e-3, "E* ± 10⁻³"),
    (RootClaim::Unique, center - 1e-9, center + 1e-9, "E* ± 10⁻⁹"),
    (
        RootClaim::Unique,
        center - 1e-15,
        center + 1e-15,
        "E* ± 10⁻¹⁵",
    ),
    (RootClaim::Unique, center, center, "E* ± 0"),
    (
        RootClaim::Empty,
        center + 1e-2,
        center + 1e-1,
        "E* + [10⁻², 10⁻¹]",
    ),
    (RootClaim::Empty, center - 1e-3, center + 1e-3, "E* ± 10⁻³"),
];
let certificates = proposals
    .iter()
    .map(|&(claim, lo, hi, _)| {
        let region = ExtendedInterval::new(Some(lo), Some(hi))?;
        let center = scioxide_core::checked_midpoint(lo, hi)?;
        Ok(ScalarRootCertificate {
            region,
            center,
            claim,
        })
    })
    .collect::<Result<Vec<_>, SciOxideError>>()
    .map_err(|e| e.to_string())?;
// 4. 検証器は設定を所有する。一括検証の要求から作業領域を事前確保する（丸めを使わない）。
let options = ValidatedOptions::new().with_byte_limit(1 << 20);
let verifier = ScalarRootVerifier::new(&value, &derivative, options);
let request = ProofRequest::proofs(certificates.len());
let requirements = verifier
    .workspace_requirements(request)
    .map_err(|e| e.to_string())?;
let mut workspace = verifier.workspace(request).map_err(|e| e.to_string())?;
emit_table_with_columns!(
    ["作業領域の要求", "値"],
    vec![
        ("独立な証明の数 [個]", requirements.proofs()),
        (
            "新しい作業領域が確保する量 [B]",
            requirements.required_bytes()
        ),
        ("保持できる量の上限 [B]", requirements.byte_limit()),
    ],
    metadata = serde_json::json!({"id":"table-0"})
);
// 5. 一括検証。レポートの欄は NotRun のレポートで初期化し、検証が上書きする。
let mut proof = || -> Result<_, SciOxideError> {
    let mut reports = vec![RootVerificationReport::not_run(); certificates.len()];
    let scope = ExecutionContext::sequential();
    verifier.verify_batch_into(&certificates, &mut reports, &mut workspace, scope)?;
    Ok(reports)
};
let reports = match proof() {
    Ok(reports) => Some(reports),
    Err(error) if unsupported(&error) => None,
    Err(error) => return Err(error.to_string()),
};
// 6. 判定はレポートの status、確立した事実は evidence で読む
let rows: Vec<_> = proposals
    .iter()
    .enumerate()
    .map(|(index, &(claim, lo, hi, label))| {
        let (status, evidence) = match &reports {
            Some(reports) => (
                format!("{:?}", reports[index].status()),
                reports[index]
                    .evidence()
                    .map_or("なし".to_string(), |fact| format!("{fact:?}")),
            ),
            None => ("未実行".to_string(), "未実行".to_string()),
        };
        (format!("{claim:?}"), label, lo, hi, status, evidence)
    })
    .collect();
emit_table_with_columns!(
    [
        "主張",
        "区間",
        "下端 [rad]",
        "上端 [rad]",
        "判定",
        "確立した事実"
    ],
    rows,
    metadata = serde_json::json!({"id":"table-1"})
);
emit_outcome!(match &reports {
    None => Outcome::unavailable("この環境では証明計算は未実行"),
    Some(reports) => Outcome::satisfied("各主張の検証を実行した。成否は表の判定を参照する")
        .diagnostic("主張の数", reports.len() as f64, None),
});
```

x86_64 のネイティブ環境では、既定値で $E^* \approx 1.786141603274$ rad になり、全域と半幅 $10^{-3}$・$10^{-9}$・$10^{-15}$ の一意性がどれも `Proved` になる。
$f'(E) = 1 - e\cos E \ge 1 - e > 0$ なので $f$ は単調増加である。
全域 $[0, 2\pi]$ では両端の $f$ の符号が逆で根の存在が示され、$0 \notin f'(X)$ から一意性が従うので、根の位置を絞らずに証明できる。
半幅 0 の点の区間では $f(E^*)$ の包含区間が 0 を含むが零そのものではなく、端点で符号が変わることも Newton 写像の包含も起こらないので、`Inconclusive(Reason(Inequality))` になる。
これは主張の否定ではない。
$[E^* + 10^{-2}, E^* + 10^{-1}]$ での不在は `Proved` になり、根を含む区間での不在の主張は `Refuted` で、確立した事実は `Unique` である。
作業領域の要求は証明 7 個、新しい作業領域の確保量 6664 B（x86_64 での値。
wasm32 では `usize` が 32 ビットなので 3752 B になる）、上限 1048576 B である。

$e = 0$ では $f$ が厳密に一次式になり、$f(E^*)$ の包含区間が厳密に零になるので、$E^* = M$ の点の区間でも一意性が `Proved` になる。
$M = 6.28$ では $E^* \approx 6.275222$ rad の 1 ulp が約 $8.9 \times 10^{-16}$ で、半幅 $10^{-15}$ は約 1 ulp にあたる。
端点での $f$ の値は丸めの幅に埋もれて符号が決まらず、この区間の一意性は `Inconclusive(Reason(Inequality))` になる。
$e = 0.99$、$M = 0$ では根が $E^* = 0$ にあり点の区間でも証明できるが、$[0.01, 0.1]$ での不在は `Inconclusive(Reason(Inequality))` になる。
この区間で $E - 0.99\sin E$ の真の値は正だが、$E$ が二回現れる依存性で値を含む区間が 0 を含み、$f'(X) \approx [0.010, 0.015]$ が小さいので Newton 写像 $N(X) \approx [-0.002, 0.016]$ も $X$ と交わり、不在を示せない。


## まとめ

狭義単調性と端点の符号は、根の存在と一意性を分けて示す。
近似根を含むだけの点区間では、一意性を証明できるとは限らない。
[根の数理](../../algorithms/root-certificates/)で判定条件を確認できる。
